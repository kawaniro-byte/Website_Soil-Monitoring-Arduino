ABOUT THE PROJECT

Why Soil Moisture Monitoring Matters

Smarter Plant Care Through Technology

Maintaining the correct soil moisture level is one of the most important factors in ensuring healthy plant growth. Too much water can cause root rot, while too little water can lead to dehydration and poor crop development.

This Soil Moisture Monitoring System uses an Arduino Uno, a Soil Moisture Sensor, an LM393 Comparator Module, LEDs, and an I2C LCD Display to continuously monitor soil moisture levels.

The system provides real-time moisture readings, displays the moisture percentage on the LCD, and visually indicates whether the soil is wet or dry using colored LEDs. This allows users to make informed decisions about when to water their plants.

Understanding the Moisture Levels

The system converts the sensor reading into a moisture percentage and assigns a corresponding soil condition.

Start the System

0%

Prepare water and begin monitoring the soil.

Very Dry

1% – 20%

The soil is very dry and needs water.

Dry

21% – 40%

The soil is dry. Watering will soon be needed.

Humid

41% – 59%

The soil has a good moisture condition.

Wet

60% – 80%

The soil has sufficient moisture and is considered healthy.

Very Wet

81% – 100%

The soil may be overwatered.

SOURCE CODE

The Code Behind the System

The Arduino program processes the sensor readings, determines the soil condition, and controls the LCD display and LED indicators.

soil_monitor.ino
C++ / Arduino 
#include <Wire.h>
#include <hd44780.h>
#include <hd44780ioClass/hd44780_I2Cexp.h>

hd44780_I2Cexp lcd;

//==============================
// Pins
//==============================
const int moisturePin = A0;
const int digitalPin  = 2;

const int redLED = 13;
const int yellowLED = 8;
const int greenLED = 7;

//==============================
// Calibration
//==============================
const int dryValue = 903;
const int wetValue = 350;

//==============================
// Timing
//==============================
const unsigned long screenInterval = 3000;
const unsigned long stableDelay = 10000;

//==============================
// Variables
//==============================
unsigned long previousScreenMillis = 0;
unsigned long lastChangeMillis = 0;

bool screenOneActive = true;

bool displayLocked = false;
bool lockScreen1 = false;
bool lockScreen2 = false;

int moistureValue = 0;
int moisturePercent = 0;
int lastMoisturePercent = -1;

String status = "";
String advice = "";

//==============================
// LCD Icons
//==============================

// Plant
byte plant[8] =
{
    B00100,
    B01110,
    B10101,
    B00100,
    B00100,
    B01110,
    B00100,
    B00000
};

// Water Drop
byte water[8] =
{
    B00100,
    B00100,
    B01110,
    B11111,
    B11111,
    B11111,
    B01110,
    B00000
};

//==============================
// Function Prototypes
//==============================
void startupScreen();
void loadingScreen();

void readSensor();
void updateStatus();
void updateLEDs();
void updateDisplayMode();

void showScreen1();
void showScreen2();
void updateLCD();

void setup()
{
    Serial.begin(9600);

    pinMode(redLED, OUTPUT);
    pinMode(yellowLED, OUTPUT);
    pinMode(greenLED, OUTPUT);

    pinMode(digitalPin, INPUT);

    lcd.begin(16, 2);

    lcd.createChar(0, plant);
    lcd.createChar(1, water);

    startupScreen();
    loadingScreen();
}

void startupScreen()
{
    lcd.clear();

    lcd.setCursor(0, 0);
    lcd.write(byte(0));
    lcd.print(" Smart Soil");

    lcd.setCursor(0, 1);
    lcd.write(byte(1));
    lcd.print(" Monitor");

    delay(2500);
}

void loadingScreen()
{
    lcd.clear();

    lcd.setCursor(0, 0);
    lcd.print("Initializing");

    lcd.setCursor(0, 1);

    for(int i = 0; i < 16; i++)
    {
        lcd.write(byte(255));
        delay(120);
    }

    delay(600);

    lcd.clear();
}

//----------------------------------------------------
// Read Soil Moisture Sensor
//----------------------------------------------------
void readSensor()
{
    long total = 0;

    // Average 10 readings for stable values
    for(int i = 0; i < 10; i++)
    {
        total += analogRead(moisturePin);
        delay(5);
    }

    moistureValue = total / 10;

    moisturePercent =
        map(moistureValue, dryValue, wetValue, 0, 100);

    moisturePercent =
        constrain(moisturePercent, 0, 100);
}

//----------------------------------------------------
// Determine Soil Status
//----------------------------------------------------
void updateStatus()
{
    if(moisturePercent == 0)
    {
        status = "Start the System.";
        advice = "Prepare Water";
    }
    else if(moisturePercent >= 1 &&
            moisturePercent <= 20)
    {
        status = "Very Dry";
        advice = "Need water";
    }
    else if(moisturePercent <= 40)
    {
        status = "Dry";
        advice = "Water Soon";
    }
    else if(moisturePercent <= 59)
    {
        status = "Humid";
        advice = "Good Cond.";
    }
    else if(moisturePercent <= 80)
    {
        status = "Wet";
        advice = "Healthy";
    }
    else
    {
        status = "Very Wet";
        advice = "Overwater";
    }
}

//----------------------------------------------------
// LED Control
//----------------------------------------------------
void updateLEDs()
{
    digitalWrite(redLED, LOW);
    digitalWrite(yellowLED, LOW);
    digitalWrite(greenLED, LOW);

    if(moisturePercent <= 40)
    {
        digitalWrite(redLED, HIGH);
    }
    else if(moisturePercent <= 59)
    {
        digitalWrite(yellowLED, HIGH);
    }
    else
    {
        digitalWrite(greenLED, HIGH);
    }
}

//----------------------------------------------------
// Smart Display Mode
//----------------------------------------------------
void updateDisplayMode()
{
    // Detect moisture change
    if(moisturePercent != lastMoisturePercent)
    {
        lastMoisturePercent = moisturePercent;

        lastChangeMillis = millis();

        displayLocked = false;
        lockScreen1 = false;
        lockScreen2 = false;
    }

    // Lock after stable for 10 seconds
    if(!displayLocked &&
       millis() - lastChangeMillis >= stableDelay)
    {
        displayLocked = true;

        lockScreen1 = false;
        lockScreen2 = false;

        if(moisturePercent == 0)
        {
            lockScreen1 = true;
        }
        else if(moisturePercent >= 10)
        {
            lockScreen2 = true;
        }
    }

    // Serial Monitor
    Serial.print("Analog: ");
    Serial.print(moistureValue);

    Serial.print("  Moisture: ");
    Serial.print(moisturePercent);
    Serial.print("%");

    Serial.print("  Status: ");
    Serial.print(status);

    Serial.print("  Locked: ");

    if(displayLocked)
        Serial.println("YES");
    else
        Serial.println("NO");
}

//----------------------------------------------------
// Screen 1
//----------------------------------------------------
void showScreen1()
{
    lcd.setCursor(0, 0);
    lcd.print("Soil Monitor   ");

    lcd.setCursor(0, 1);

    lcd.print("Moist:");

    if(moisturePercent < 10)
        lcd.print(" ");

    if(moisturePercent < 100)
        lcd.print(" ");

    lcd.print(moisturePercent);
    lcd.print("%   ");
}

//----------------------------------------------------
// Screen 2
//----------------------------------------------------
void showScreen2()
{
    lcd.setCursor(0, 0);
    lcd.print("                ");

    lcd.setCursor(0, 0);

    lcd.print("Status:");
    lcd.print(status);

    lcd.setCursor(0, 1);
    lcd.print("                ");

    lcd.setCursor(0, 1);

    lcd.print(advice);
}

//----------------------------------------------------
// LCD Manager
//----------------------------------------------------
void updateLCD()
{
    static bool lastScreen = true;
    static bool lastLocked = false;

    if(displayLocked)
    {
        if(lastLocked == false)
        {
            lcd.clear();
            lastLocked = true;
        }

        if(lockScreen1)
        {
            showScreen1();
        }
        else if(lockScreen2)
        {
            showScreen2();
        }

        return;
    }

    lastLocked = false;

    if(millis() - previousScreenMillis >= screenInterval)
    {
        previousScreenMillis = millis();

        screenOneActive = !screenOneActive;

        lcd.clear();
    }

    if(screenOneActive)
        showScreen1();
    else
        showScreen2();
}

void loop()
{
    readSensor();

    updateStatus();

    updateLEDs();

    updateDisplayMode();

    updateLCD();
}

How the Program Works

The program reads the analog value from the soil moisture sensor and converts it into a percentage. The percentage is then used to determine the current condition of the soil.

Sensor Reading

Reads the moisture level through analog pin A0.

Data Processing

Converts the sensor reading into a moisture percentage.

Status Detection

Determines whether the soil is very dry, dry, humid, wet, or very wet.

User Feedback

Sends the results to the LCD and LED indicators.
